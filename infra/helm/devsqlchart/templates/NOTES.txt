1. Get the application URL by running these commands:
{{- range $service := .Values.service.instances }}
{{- if eq $service.type "NodePort" }}
  export NODE_PORT=$(kubectl get --namespace {{ $.Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ $service.name }})
  export NODE_IP=$(kubectl get nodes --namespace {{ $.Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if eq $service.type "ClusterIP" }}
  kubectl port-forward --namespace {{ $.Release.Namespace }} svc/{{ $service.name }} {{ $service.port }}:{{ $service.port }}
  echo "Visit http://127.0.0.1:{{ $service.port }} to access {{ $service.name }}"
{{- end }}
{{- end }}
